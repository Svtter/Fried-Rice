{{- /* Series navigation box shown on article pages that belong to a series. Displays the optional series cover, the series title (linking to the series page), an ordered list of all parts with the current article highlighted, and prev/next links with adjacent part titles. Ordering matches the series landing page: newest first (weight desc, date desc fallback). */ -}} {{- $enabled := default true .Site.Params.article.series.enabled -}} {{- if and $enabled .Params.series -}} {{- $seriesName := index .Params.series 0 -}} {{- $series := .Site.GetPage (printf "/series/%s" $seriesName) -}} {{- with $series -}} {{- $current := $ -}} {{- $currentPermalink := $current.RelPermalink -}} {{- $ordered := sort (sort .Pages "Date" "asc") "Params.weight" "asc" -}} {{- $currentIndex := 0 -}} {{- range $i, $page := $ordered -}} {{- if eq $page.RelPermalink $currentPermalink -}} {{- $currentIndex = $i -}} {{- end -}} {{- end -}} {{- $prev := false -}} {{- $next := false -}} {{- if gt $currentIndex 0 -}} {{- $prev = index $ordered (sub $currentIndex 1) -}} {{- end -}} {{- if lt $currentIndex (sub (len $ordered) 1) -}} {{- $next = index $ordered (add $currentIndex 1) -}} {{- end -}}
{{- $image := partialCached "helper/image" (dict "Image" .Params.image "Resources" .Resources "Context" .) .RelPermalink "series-box" -}} {{- if $image -}} {{- if $image.Resource -}} {{- $Permalink := $image.Resource.RelPermalink -}} {{- $Width := $image.Resource.Width -}} {{- $Height := $image.Resource.Height -}} {{- if and (default true .Site.Params.imageProcessing.cover.enabled) (reflect.IsImageResourceProcessable $image.Resource) -}} {{- $cover := $image.Resource.Fill "800x300" -}} {{- $Permalink = $cover.RelPermalink -}} {{- $Width = $cover.Width -}} {{- $Height = $cover.Height -}} {{- end -}} {{- else -}} {{- end -}} {{- end -}}

{{ partial "helper/icon" "series" }} {{ .Title }}

    {{- range $i, $page := $ordered -}}
  1. {{ T "article.series.part" (add $i 1) }} {{ $page.Title }}
    2. {{- end -}}
{{- if or $prev $next -}}
{{- with $prev -}} ← {{ T "article.series.prev" }} {{ .Title }} {{- end -}} {{- with $next -}} {{ T "article.series.next" }} → {{ .Title }} {{- end -}}
{{- end -}}
{{- end -}} {{- end -}}
